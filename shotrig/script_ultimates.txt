# The ultimates: sneak + left-click on every weapon, photographed mid-animation.
# Player at 0 -60 0 facing south; three husks around as targets so the area effects show.
wait 60
cmd gamemode survival
cmd gamerule do_daylight_cycle false
cmd gamerule do_weather_cycle false
cmd gamerule spawn_mobs false
cmd gamerule mob_griefing false
cmd time set 6000
cmd weather clear
cmd kill @e[type=!player]
cmd tp @s 0.5 -60 0.5 0 10
cmd clear @s
cmd effect give @s minecraft:resistance 9999 4 true
cmd effect give @s minecraft:saturation 9999 4 true
wait 20
look 0 10
wait 10

# ---- Crimson Nova
cmd kill @e[type=!player]
cmd tp @s 0.5 -60 0.5 0 10
cmd summon husk 0.5 -60 3.5 {NoAI:1b,Silent:1b,PersistenceRequired:1b,attributes:[{id:"minecraft:max_health",base:200}],Health:200f}
cmd summon husk -2.5 -60 2.0 {NoAI:1b,Silent:1b,PersistenceRequired:1b,attributes:[{id:"minecraft:max_health",base:200}],Health:200f}
cmd summon husk 3.5 -60 1.5 {NoAI:1b,Silent:1b,PersistenceRequired:1b,attributes:[{id:"minecraft:max_health",base:200}],Health:200f}
cmd clear @s
cmd customweapon give @s bloodletter
wait 20
hotbar 0
look 0 10
camera first
wait 10
shot 01_bloodletter_a_idle
camera back
wait 5
sneak on
wait 3
attack
wait 2
shot 01_bloodletter_b_ult_t2
wait 3
shot 01_bloodletter_c_ult_t5
wait 3
shot 01_bloodletter_d_ult_t8
wait 4
shot 01_bloodletter_e_t12_landing
wait 6
shot 01_bloodletter_e_after
camera first
wait 2
shot 01_bloodletter_f_after_firstperson
sneak off
wait 30
